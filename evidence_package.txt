def estimate_encrypted_payload_len(raw_len):
    padded_len = ((raw_len // 16) + 1) * 16
    return AES_SALT_SIZE + AES_IV_SIZE + padded_len

def check_image_for_message(img_path, delim):
    try:
        img = Image.open(img_path).convert("RGB")
        arr = np.array(img)
        flat = arr.reshape(-1) & 1
        usable_len = (flat.size // 8) * 8
        if usable_len == 0: return False, False
        raw_bytes = np.packbits(flat[:usable_len]).tobytes()
        delim_bytes = delim.encode("utf-8")
        idx_stealth = raw_bytes.find(STEALTH_HEADER)
        idx_legacy = raw_bytes.find(LEGACY_HEADER)
        if idx_stealth != -1:
            idx = idx_stealth; header_len = len(STEALTH_HEADER)
        elif idx_legacy != -1:
            idx = idx_legacy; header_len = len(LEGACY_HEADER)
        else:
            return False, False
        payload_start = idx + header_len
        d_idx = raw_bytes[payload_start:].find(delim_bytes)
        if d_idx == -1: return False, False
        payload_bytes = raw_bytes[payload_start : payload_start + d_idx]
        try:
            payload_bytes.decode("utf-8")
            return True, False
        except UnicodeDecodeError:
            return True, True
    except Exception:
        return False, False

    def _update_watermark(self):
        if not hasattr(self, 'watermark') or not hasattr(self, 'ws_frame'): return
        text = self.watermark.text()
        if not text: return
        self.watermark.updateGeometry()
        self.ws_frame.updateGeometry()
        if self.ws_frame.layout(): self.ws_frame.layout().update()
        frame_rect = self.ws_frame.rect()
        margins = self.ws_frame.layout().contentsMargins() if self.ws_frame.layout() else QMargins(15, 15, 15, 15)
        controls_height = 60
        target_w = frame_rect.width() - margins.left() - margins.right() - 60
        target_h = frame_rect.height() - margins.top() - margins.bottom() - controls_height - 60
        if target_w <= 0 or target_h <= 0:
            window_rect = self.rect()
            target_w = max(200, window_rect.width() * 0.8)
            target_h = max(100, window_rect.height() * 0.5)
            if target_w <= 0 or target_h <= 0: target_w = 800; target_h = 400
        ref_size = WATERMARK_REF_SIZE
        font = QFont("Segoe UI", ref_size, QFont.Bold)
        font.setPixelSize(ref_size)
        font.setKerning(False)
        metrics = QFontMetrics(font)
        rect = metrics.boundingRect(text)
        if rect.width() == 0 or rect.height() == 0: return
        scale_w = target_w / rect.width() if rect.width() > 0 else 1.0
        scale_h = target_h / rect.height() if rect.height() > 0 else 1.0
        scale = min(scale_w, scale_h)
        scale = max(WATERMARK_MIN_SCALE, min(WATERMARK_MAX_SCALE, scale))
        final_size = int(ref_size * scale)
        final_size = max(WATERMARK_MIN_SIZE, min(final_size, WATERMARK_MAX_SIZE))
        font.setPixelSize(final_size)
        self.watermark.setFont(font)
        self.watermark.setStyleSheet(f"color: rgba(255, 255, 255, 0.18); background: transparent; font-size: {final_size}px;")

    def _extract_metadata(self, path):
        if not path or not os.path.exists(path): return "<i>No image loaded</i>"
        try:
            img = Image.open(path)
            exif = img.getexif()
            lines = [
                f"<b>File:</b> {os.path.basename(path)}",
                f"<b>Size:</b> {self._format_size(os.path.getsize(path))}",
                f"<b>Dimensions:</b> {img.width} Ã— {img.height} px",
                f"<b>Mode:</b> {img.mode}",
                f"<b>Format:</b> {img.format or 'Unknown'}",
                f"<b>Capacity:</b> {self._format_size(self.available_capacity_bits // 8)} ({(self.available_capacity_bits // 8):,} chars)",
                "<hr style='border-color:#444'>",
                "<b>EXIF / Metadata:</b>"
            ]
            if exif:
                for tag_id, val in exif.items():
                    try:
                        tag_name = html.escape(str(ExifTags.TAGS.get(tag_id, str(tag_id))))
                        v = html.escape(str(val))
                        lines.append(f"â€¢ {tag_name}: {v[:60]}{'...' if len(v) > 60 else ''}")
                    except Exception as e:
                        logger.debug(f"Skipping unreadable EXIF tag {tag_id}: {e}")
                        continue
            else:
                lines.append("â€¢ No EXIF data found.")
            return "<br>".join(lines)
        except Exception as e:
            return f"<i>Error: {str(e)}</i>"

import hashlib
import itertools
import secrets
import numpy as np
from PIL import Image
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import padding
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.hazmat.primitives import hashes

STEALTH_HEADER = hashlib.sha256(b"STEGO_MAGIC").digest()
LEGACY_HEADER = b"STEGO_MAGIC"
DUAL_HEADER = hashlib.sha256(b"STEGO_DUAL_MAGIC").digest()
DELIMITER = "\x00\x00\x03"

AES_KEY_SIZE = 32
AES_SALT_SIZE = 16
AES_IV_SIZE = 16
PBKDF2_ITERATIONS = 100000

def _encrypt_bytes_aes(data, password):
    if not password: return data
    salt = secrets.token_bytes(AES_SALT_SIZE)
    iv = secrets.token_bytes(AES_IV_SIZE)
    kdf = PBKDF2HMAC(algorithm=hashes.SHA256(), length=AES_KEY_SIZE, salt=salt, iterations=PBKDF2_ITERATIONS, backend=default_backend())
    key = kdf.derive(password.encode())
    padder = padding.PKCS7(128).padder()
    padded_data = padder.update(data) + padder.finalize()
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    encryptor = cipher.encryptor()
    encrypted_data = encryptor.update(padded_data) + encryptor.finalize()
    return salt + iv + encrypted_data

def _decrypt_bytes_aes(encrypted_data, password):
    if not password: return encrypted_data
    if len(encrypted_data) < AES_SALT_SIZE + AES_IV_SIZE: return None
    salt = encrypted_data[:AES_SALT_SIZE]
    iv = encrypted_data[AES_SALT_SIZE:AES_SALT_SIZE + AES_IV_SIZE]
    encrypted = encrypted_data[AES_SALT_SIZE + AES_IV_SIZE:]
    kdf = PBKDF2HMAC(algorithm=hashes.SHA256(), length=AES_KEY_SIZE, salt=salt, iterations=PBKDF2_ITERATIONS, backend=default_backend())
    key = kdf.derive(password.encode())
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    decryptor = cipher.decryptor()
    try:
        padded_data = decryptor.update(encrypted) + decryptor.finalize()
        unpadder = padding.PKCS7(128).unpadder()
        data = unpadder.update(padded_data) + unpadder.finalize()
        return data
    except Exception:
        return None

def _encrypt_bytes(data, password):
    if not password: return data
    key = hashlib.sha256(password.encode()).digest()
    return bytes(b ^ k for b, k in zip(data, itertools.cycle(key)))

def _embed_bits_and_save(img_path, full_data, save_path, out_format="PNG"):
    img = Image.open(img_path)
    exif_data, icc_data, dpi_data = img.info.get("exif"), img.info.get("icc_profile"), img.info.get("dpi")
    img_rgb = img.convert("RGB")
    arr = np.array(img_rgb)
    h, w, c = arr.shape
    flat = arr.reshape(-1)
    bits = np.unpackbits(np.frombuffer(full_data, dtype=np.uint8))
    if bits.size > flat.size: raise ValueError("Message exceeds image capacity.")
    flat[:bits.size] = (flat[:bits.size] & 0xFE) | bits
    new_img = Image.fromarray(flat.reshape(h, w, c).astype(np.uint8), "RGB")
    out_format = out_format.upper()
    if out_format == "BMP": kwargs = {"format": "BMP"}
    elif out_format == "TIFF":
        kwargs = {"format": "TIFF"}
        if icc_data: kwargs["icc_profile"] = icc_data
    elif out_format == "WEBP":
        kwargs = {"format": "WEBP", "lossless": True}
        if icc_data: kwargs["icc_profile"] = icc_data
    else:
        kwargs = {"format": "PNG"}
        if exif_data: kwargs["exif"] = exif_data
        if icc_data: kwargs["icc_profile"] = icc_data
        if dpi_data: kwargs["dpi"] = dpi_data
    new_img.save(save_path, **kwargs)

def encode_image(img_path, msg, save_path, header_bytes, delim, out_format="PNG", password=""):
    payload = msg.encode("utf-8")
    if password: payload = _encrypt_bytes_aes(payload, password)
    full_data = header_bytes + payload + delim.encode("utf-8")
    _embed_bits_and_save(img_path, full_data, save_path, out_format)

def encode_dual_image(img_path, msg1, msg2, save_path, dual_header_bytes, delim, out_format="PNG", pass1="", pass2=""):
    payload1 = msg1.encode("utf-8")
    if pass1: payload1 = _encrypt_bytes_aes(payload1, pass1)
    payload2 = _encrypt_bytes_aes(msg2.encode("utf-8"), pass2)
    delim_bytes = delim.encode("utf-8")
    full_data = dual_header_bytes + payload1 + delim_bytes + payload2 + delim_bytes
    _embed_bits_and_save(img_path, full_data, save_path, out_format)

def decode_image(img_path, delim, password=""):
    img = Image.open(img_path).convert("RGB")
    arr = np.array(img)
    flat = arr.reshape(-1) & 1
    usable_len = (flat.size // 8) * 8
    if usable_len == 0: return "", False
    raw_bytes = np.packbits(flat[:usable_len]).tobytes()
    delim_bytes = delim.encode("utf-8")
    idx_stealth = raw_bytes.find(STEALTH_HEADER)
    idx_legacy = raw_bytes.find(LEGACY_HEADER)
    if idx_stealth != -1:
        idx = idx_stealth; header_len = len(STEALTH_HEADER)
    elif idx_legacy != -1:
        idx = idx_legacy; header_len = len(LEGACY_HEADER)
    else:
        return "", False
    payload_start = idx + header_len
    d_idx = raw_bytes[payload_start:].find(delim_bytes)
    if d_idx == -1: return "", False
    payload_bytes = raw_bytes[payload_start : payload_start + d_idx]
    if password:
        try:
            decrypted_bytes = _decrypt_bytes_aes(payload_bytes, password)
            if decrypted_bytes is not None:
                text = decrypted_bytes.decode("utf-8")
                return text, True
        except (UnicodeDecodeError, ValueError, TypeError):
            pass
        try:
            decrypted_bytes = _encrypt_bytes(payload_bytes, password)
            text = decrypted_bytes.decode("utf-8")
            return text, True
        except (UnicodeDecodeError, ValueError, TypeError):
            pass
        try:
            text = payload_bytes.decode("utf-8")
            return text, True
        except UnicodeDecodeError:
            return "", False
    try:
        text = payload_bytes.decode("utf-8")
        return text, False
    except UnicodeDecodeError:
        return "", False

def _extract_dual_layers(img_path, delim):
    raw_bytes = _extract_raw_lsb_bytes(img_path)
    if not raw_bytes:
        return None, None
    idx = raw_bytes.find(DUAL_HEADER)
    if idx == -1:
        return None, None
    payload_start = idx + len(DUAL_HEADER)
    delim_bytes = delim.encode("utf-8")
    d1_idx = raw_bytes[payload_start:].find(delim_bytes)
    if d1_idx == -1:
        return None, None
    layer1_end = payload_start + d1_idx
    layer1_bytes = raw_bytes[payload_start:layer1_end]
    layer2_start = layer1_end + len(delim_bytes)
    d2_idx = raw_bytes[layer2_start:].find(delim_bytes)
    if d2_idx == -1:
        return layer1_bytes, None
    layer2_end = layer2_start + d2_idx
    layer2_bytes = raw_bytes[layer2_start:layer2_end]
    return layer1_bytes, layer2_bytes

        except (UnicodeDecodeError, ValueError, TypeError):
            pass
        try:
            return payload_bytes.decode("utf-8"), False
        except UnicodeDecodeError:

    try:
        return payload_bytes.decode("utf-8"), False

_KEYBOARD_ROWS = ["qwertyuiop", "asdfghjkl", "zxcvbnm", "1234567890"]

def _has_repeated_char_run(password, run_length=3):
    count = 1
    for i in range(1, len(password)):
        if password[i] == password[i - 1]:
            count += 1
            if count >= run_length:
                return True
        else:
            count = 1
    return False

def _has_sequential_run(password, run_length=3):
    if len(password) < run_length:
        return False
    for i in range(len(password) - run_length + 1):
        chunk = password[i:i + run_length]
        ascending = all(ord(chunk[j + 1]) - ord(chunk[j]) == 1 for j in range(len(chunk) - 1))
        descending = all(ord(chunk[j]) - ord(chunk[j + 1]) == 1 for j in range(len(chunk) - 1))
        if ascending or descending:
            return True
    return False

def _has_keyboard_pattern(password, min_len=4):
    lowered = password.lower()
    for row in _KEYBOARD_ROWS:
        for i in range(len(row) - min_len + 1):
            chunk = row[i:i + min_len]
            if chunk in lowered or chunk[::-1] in lowered:
                return True
    return False

def _extract_alpha_runs(password):
    runs, current = [], ""
    for c in password:
        if c.isalpha():
            current += c
        else:
            if current: runs.append(current)
            current = ""
    if current: runs.append(current)
    return runs

def _has_common_word(password):
    return any(run.lower() in _COMMON_WEAK_WORDS for run in _extract_alpha_runs(password) if len(run) >= 4)

def _has_pattern_weakness(password):
    return (_has_common_word(password) or _has_keyboard_pattern(password)
            or _has_repeated_char_run(password) or _has_sequential_run(password))

def estimate_password_strength(password):
    if not password:
        return 0
    length_score = min(len(password), 20) / 20 * 60
    upper = sum(1 for c in password if c.isupper())
    digit = sum(1 for c in password if c.isdigit())
    symbol = sum(1 for c in password if not c.isalnum())
    diversity_score = sum(min(count, 3) / 3 * (40 / 3) for count in (upper, digit, symbol))
    score = length_score + diversity_score
    if _has_pattern_weakness(password):
        score = min(score, 30)
    return min(int(round(score)), 100)

def _password_strength_tier(score):
    if score <= 0: return 0
    if score < 20: return 1
    if score < 40: return 2
    if score < 60: return 3
    if score < 80: return 4
    return 5

from PyQt5.QtCore import QThread, pyqtSignal

logger = logging.getLogger("Stegano")


class EncodeWorker(QThread):
    encoding_finished = pyqtSignal(str)
    error = pyqtSignal(str)
    def __init__(self, target_fn, args, parent=None):
        super().__init__(parent)
        self.target_fn = target_fn
        self.args = args
    def run(self):
        try:
            self.target_fn(*self.args)
            self.encoding_finished.emit("OK")
        except Exception as e:
            logger.exception("Encoding failed")
            self.error.emit(str(e))

class DecodeWorker(QThread):
    decoding_finished = pyqtSignal(str, bool, int, str)
    error = pyqtSignal(str)
    def __init__(self, path, delimiter, pass1, pass2, parent=None):
        super().__init__(parent)
        self.path = path
        self.delimiter = delimiter
        self.pass1 = pass1
        self.pass2 = pass2
    def run(self):
        try:
            from steganography_utils import detect_message_format, decode_image, decode_dual_image
            fmt = detect_message_format(self.path, self.delimiter)
            if fmt == 'dual':
                msg, layer, was_encrypted, layer1_msg = decode_dual_image(self.path, self.delimiter, self.pass1, self.pass2)
                self.decoding_finished.emit(msg, was_encrypted, layer, layer1_msg)
            else:
                msg, is_encrypted = decode_image(self.path, self.delimiter, self.pass1)
                layer = 1 if msg else 0
                self.decoding_finished.emit(msg, is_encrypted, layer, msg)
        except Exception as e:
            logger.exception("Decoding failed")
            self.error.emit(str(e))

class ImageLoadWorker(QThread):
    loaded = pyqtSignal(dict)
    error = pyqtSignal(str)
    def __init__(self, path, delimiter, parent=None):
        super().__init__(parent)
        self.path = path
        self.delimiter = delimiter
    def run(self):
        try:
            from PIL import Image
            from steganography_utils import detect_message_format, decode_image, decode_dual_image, check_image_for_message
            img = Image.open(self.path).convert("RGB")
            capacity_bits = img.size[0] * img.size[1] * 3
            fmt = detect_message_format(self.path, self.delimiter)
            has_message = fmt in ('single', 'dual')
            is_encrypted = False
            message = ""
            if fmt == 'dual':
                msg, layer, was_encrypted, layer1_msg = decode_dual_image(self.path, self.delimiter, "", "")
                if layer > 0:
                    message = msg
                    is_encrypted = was_encrypted
                else:
                    is_encrypted = True
            elif fmt == 'single':
                has_message, is_encrypted = check_image_for_message(self.path, self.delimiter)
                if has_message and not is_encrypted:
                    message, _ = decode_image(self.path, self.delimiter, "")
            self.loaded.emit({
                'capacity_bits': capacity_bits,
                'has_message': has_message,
                'is_encrypted': is_encrypted,
                'message': message,
            })
        except Exception as e:
            logger.exception("Image load/scan failed")
            self.error.emit(str(e))

import os
import wave
import numpy as np
from PyQt5.QtMultimedia import QSoundEffect, QMediaPlayer, QMediaContent
from PyQt5.QtCore import QUrl, QTimer
import logging

logger = logging.getLogger("Stegano")

AUDIO_SAMPLE_RATE = 44100
AUDIO_FADE_DURATION = 0.05

class AudioManager:
    def __init__(self):
        self.sfx = {}
        self.music_player = QMediaPlayer()
        self.music_enabled = True
        self.sfx_enabled = True
        self.base_dir = os.path.dirname(os.path.abspath(__file__))
        self._ensure_audio_files()
        self._load_sounds()
        self.set_music("idle")
        self.music_player.mediaStatusChanged.connect(self._handle_media_status)
    def _ensure_audio_files(self):
        sfx_dir = os.path.join(self.base_dir, "audio", "sfx")
        music_dir = os.path.join(self.base_dir, "audio", "music")
        os.makedirs(sfx_dir, exist_ok=True)
        os.makedirs(music_dir, exist_ok=True)
        sfx_map = {'click.wav': (800, 0.08, 0.3), 'action.wav': (600, 0.15, 0.4), 'success.wav': (880, 0.4, 0.3), 'error.wav': (250, 0.3, 0.4)}
        music_map = {'idle.wav': (110, 12.0, 0.15)}
        try:
            for name, (freq, dur, vol) in {**sfx_map, **music_map}.items():
                folder = sfx_dir if name in sfx_map else music_dir
                path = os.path.join(folder, name)
                if not os.path.exists(path):
                    self._create_audio(path, freq, dur, vol, is_music=(name in music_map))
        except Exception as e:
            logger.warning(f"Could not generate default audio files: {e}")
    def _create_audio(self, filepath, base_freq, duration, volume, sr=AUDIO_SAMPLE_RATE, is_music=False):
        t = np.linspace(0, duration, int(sr * duration), False)
        audio = np.zeros_like(t)
        if is_music:
            for f in [base_freq, base_freq*1.25, base_freq*1.5, base_freq*2.0]:
                audio += np.sin(2 * np.pi * f * t) * 0.15
            lfo = 0.5 + 0.5 * np.sin(2 * np.pi * 0.1 * t)
            audio *= lfo
        else:
            audio = np.sin(2 * np.pi * base_freq * t) * 0.5 * np.exp(-t * 15)
        env = np.ones_like(t)
        fade = int(sr * AUDIO_FADE_DURATION)
        if fade < len(env):
            env[:fade] = np.linspace(0, 1, fade)
            env[-fade:] = np.linspace(1, 0, fade)
        audio *= env * volume
        data = (audio * 32767).astype(np.int16).tobytes()
        with wave.open(filepath, 'wb') as w:
            w.setnchannels(1)
            w.setsampwidth(2)
            w.setframerate(sr)
            w.writeframes(data)
    def _load_sounds(self):
        sfx_dir = os.path.join(self.base_dir, "audio", "sfx")
        if not os.path.exists(sfx_dir): return
        for name in ["click", "action", "success", "error"]:
            path = os.path.join(sfx_dir, f"{name}.wav")
            if os.path.exists(path):
                eff = QSoundEffect()
                eff.setSource(QUrl.fromLocalFile(path))
                eff.setVolume(0.7)
                self.sfx[name] = eff
    def play_sfx(self, name):
        if self.sfx_enabled and name in self.sfx:
            try: self.sfx[name].play()
            except Exception as e: logger.debug(f"Could not play sound effect '{name}': {e}")
    def set_music(self, mode):
        if not self.music_enabled: return
        music_dir = os.path.join(self.base_dir, "audio", "music")
        for ext in [".mp3", ".wav"]:
            path = os.path.join(music_dir, f"{mode}{ext}")
            if os.path.exists(path):
                self.music_player.stop()
                self.music_player.setMedia(QMediaContent(QUrl.fromLocalFile(path)))
                self.music_player.setVolume(60)
                self.music_player.play()
                return
    def _handle_media_status(self, state):
        if state == QMediaPlayer.EndOfMedia and self.music_enabled:
            self.music_player.setPosition(0)
            self.music_player.play()
    def set_music_volume(self, vol):
        self.music_player.setVolume(max(0, min(100, vol)))
    def toggle_music(self):
        self.music_enabled = not self.music_enabled
        if not self.music_enabled:
            self.music_player.stop()
        else:
            self.set_music("idle")
        return self.music_enabled
    def toggle_sfx(self):
        self.sfx_enabled = not self.sfx_enabled
        return self.sfx_enabled
